// Permisos que no tienen sentido para un rol
function noAplica(rol, k) {
  if (rol === 'colaborador' && k.startsWith('com.')) return 'El colaborador no tiene comisiones.';
  if (rol === 'comercial' && k === 'com.ver_equipo') return 'Un comercial no tiene equipo.';
  return '';
}
const rolBadge = r => { const x = ROLES[r] || { n: r, c: '#334155', bg: '#e2e8f0', i: 'fa-user' }; return `<span class="rb" style="background:${x.bg};color:${x.c}"><i class="fas ${x.i} me-1"></i>${esc(x.n)}</span>`; };
const iniciales = s => String(s || '?').trim().split(/\s+/).slice(0, 2).map(w => w[0]).join('').toUpperCase();
function personalizados(rol, ov) {
  let mas = 0, menos = 0;
  Object.keys(ov || {}).forEach(k => { if (PERM_NOMBRE[k] && !noAplica(rol, k) && ov[k] !== (ROL_DEF[rol] || []).includes(k)) { if (ov[k]) mas++; else menos++; } });
  return { mas, menos };
}

async function guardarUsuarioEd() {
  leerEditor(); limpiarOv();
  const nombre = (ED.nombre || '').trim(), dni = (ED.dni || '').trim(), email = (ED.email || '').trim().toLowerCase();
  const falta = m => { ED.tab = 'datos'; pintarEditor(); alert(m); };
  if (nombre.length < 3) return falta('Escribe el nombre y apellidos completos.');
  if (!ED.id && dni.length < 4) return falta('Escribe el documento de identidad.');
  if (!ED.id && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return falta('Escribe un correo electrónico válido.');
  const btn = document.getElementById('mu-guardar'); btn.disabled = true;
  const cuerpo = { nombre, dni: dni || null, rol: ED.rol, manager_id: ED.jefe || null, companias: ED.rol === 'director' ? [] : ED.companias, equipos: ED.rol === 'colaborador' ? ED.equipos : [], permisos: ED.ov };
  let msg = null, data = null;
  if (ED.id) {
    const r = await db.from('profiles').update({ ...cuerpo, activo: ED.activo }).eq('id', ED.id);
    msg = r.error && r.error.message;
  } else {
    const r = await db.functions.invoke('smooth-responder', { body: { ...cuerpo, email, companias: cuerpo.companias.join(',') } });
    data = r.data; msg = data && data.error;
    if (r.error && !msg) { try { msg = (await r.error.context.json()).error; } catch (_) { msg = r.error.message; } }
  }
  btn.disabled = false;
  if (msg) { alert('No se pudo guardar: ' + msg); return; }
  bootstrap.Modal.getInstance(document.getElementById('mUsuario')).hide();
  if (!ED.id) alert(data && data.correo ? 'Usuario creado. Se envió el correo a ' + email + '.'
    : 'Usuario creado, pero el correo NO se pudo enviar.\n\nEntrégale esta contraseña temporal por otro medio:\n' + (data && data.clave) + '\n\nDetalle del error: ' + ((data && data.detalle) || 'sin detalle'));

  await cargarEquipo(); cargarVendedores();

function ibanValido(v) {
  const s = String(v || '').replace(/\s/g, '').toUpperCase();
  if (!/^[A-Z]{2}[0-9]{2}[A-Z0-9]{11,30}$/.test(s)) return false;
  let resto = 0;
  for (const ch of s.slice(4) + s.slice(0, 4)) {
    const n = /[0-9]/.test(ch) ? ch : String(ch.charCodeAt(0) - 55);
    for (const d of n) resto = (resto * 10 + +d) % 97;
  }
  if (resto !== 1) return false;
  if (s.startsWith('ES')) {
    if (!/^ES[0-9]{22}$/.test(s)) return false;
    const w = [1, 2, 4, 8, 5, 10, 9, 7, 3, 6];
    const dc = t => { const r = 11 - [...t].reduce((a, d, i) => a + d * w[i], 0) % 11; return r === 11 ? 0 : r === 10 ? 1 : r; };
    if (dc('00' + s.slice(4, 12)) !== +s[12] || dc(s.slice(14, 24)) !== +s[13]) return false;
  }
  return true;
}
function revisarIban(el, salir) {
  const msg = document.getElementById('iban-msg'), v = el.value;
  if (!v) { el.setCustomValidity(''); msg.textContent = ''; el.classList.remove('is-invalid', 'is-valid'); return; }
  if (v.length < 24 && !salir) { el.setCustomValidity('El IBAN debe tener 24 caracteres'); msg.className = 'form-text'; msg.textContent = v.length + ' de 24 caracteres'; el.classList.remove('is-invalid', 'is-valid'); return; }
  const ok = ibanValido(v);
  el.setCustomValidity(ok ? '' : 'IBAN no válido: revisa que esté bien escrito');
  el.classList.toggle('is-valid', ok); el.classList.toggle('is-invalid', !ok);
  msg.className = 'form-text ' + (ok ? 'text-success' : 'text-danger');
  msg.textContent = ok ? 'IBAN válido' : (v.length !== 24 ? 'El IBAN debe tener 24 caracteres' : 'IBAN no válido: algún dígito está mal escrito');
}

function pasa(c) {
  const f = FILTRO; if (!f) return true;
  for (const [k, , campo] of FDATES) {
    const [a, b] = f.fechas[k], d = ymd(c[campo]);
    if ((a || b) && !d) return false;
    if (a && d < a) return false;
    if (b && d > b) return false;
  }
  if (f.estados.length && !f.estados.includes(c.estado)) return false;
  if (f.comp && c.compania !== f.comp) return false;
  if (f.solar === 'SI' && !c.tarifa_solar) return false;
  if (f.solar === 'NO' && (c.tarifa_solar || !c.cups_luz)) return false;
  for (const [k, , fn] of FTXT) { const t = f.txt[k]; if (t && !String(fn(c) || '').toLowerCase().includes(t)) return false; }
  return true;
}

function analitica() {
  const box = document.getElementById('analitica'); if (!box || !perfil) return;
  const dia = d => new Date(d).toLocaleDateString('sv-SE');
  const hoy = new Date(), hoyS = dia(hoy), ayerS = dia(new Date(hoy.getTime() - 864e5));
  const ini = corte || new Date(hoy.getFullYear(), hoy.getMonth(), 1), iniS = dia(ini);
  const nuevas = contratos.filter(c => new Date(c.created_at) >= ini);
  const delHoy = contratos.filter(c => dia(c.created_at) === hoyS);
  const nAyer = contratos.filter(c => dia(c.created_at) === ayerS).length;
  const ultima = delHoy.length ? new Date(Math.max(...delHoy.map(c => new Date(c.created_at).getTime()))).toLocaleTimeString('es-CO', { hour: '2-digit', minute: '2-digit' }) : null;
  const esBaja = c => c.estado === 'BAJA' && c.fecha_baja && c.fecha_baja >= iniS;
  const bajas = contratos.filter(esBaja).length;
  const formal = contratos.filter(c => c.estado === 'FORMALIZADO' && cuenta(c)).length;
  const pct = nuevas.length ? bajas / nuevas.length * 100 : null;
  const alcance = { comercial: 'Mis ventas', supervisor: 'Mi equipo', director: 'General', colaborador: 'Mis campañas y equipos' }[perfil.rol] || '';
  const desde = new Date(Math.max(new Date(iniS + 'T00:00:00').getTime(), hoy.getTime() - 30 * 864e5));
  const dias = [];
  for (let d = new Date(desde.getFullYear(), desde.getMonth(), desde.getDate()); dia(d) <= hoyS; d.setDate(d.getDate() + 1)) dias.push(dia(d));
  const porDia = dias.map(d => nuevas.filter(c => dia(c.created_at) === d).length);
  let acc = 0; const acum = porDia.map(v => (acc += v));
  const horas = Array(24).fill(0); nuevas.forEach(c => { horas[new Date(c.created_at).getHours()]++; });
  const maxH = Math.max(...horas), pico = maxH > 0 ? horas.indexOf(maxH) : -1;
  const porEstado = ESTADOS.map(([k, l, col]) => ({ l, col, n: contratos.filter(c => c.estado === k && cuenta(c)).length })).filter(x => x.n > 0);
  const puedeRank = ['director', 'supervisor'].includes(perfil.rol);
  let rank = [];
  if (puedeRank) {
    const m = {}, nom = c => (c.owner && (c.owner.nombre || c.owner.email)) || '—';
    nuevas.forEach(c => { const k = nom(c); (m[k] = m[k] || { v: 0, b: 0 }).v++; });
    contratos.filter(esBaja).forEach(c => { const k = nom(c); (m[k] = m[k] || { v: 0, b: 0 }).b++; });
    rank = Object.entries(m).sort((a, z) => z[1].v - a[1].v).slice(0, 10);
  }
  ANA = { dias, porDia, acum, horas, pico, maxH, porEstado, rank };
  const tabs = [['dia', 'Progreso del mes'], ['hora', 'Por hora'], ['est', 'Por estado']].concat(puedeRank ? [['rank', 'Ranking']] : []);
  if (!tabs.some(t => t[0] === ANA_TAB)) ANA_TAB = 'dia';
  const st = (t, v, s, col) => `<div class="st"><div class="st-v" style="color:${col}">${v}</div><div class="st-t">${t}</div>${s ? `<div class="st-s">${s}</div>` : ''}</div>`;
  box.innerHTML = `<div class="card-g lite">
    <div class="d-flex flex-wrap justify-content-between align-items-center gap-2 mb-2">
      <div class="sec-t">Resumen de ventas · ${alcance}</div>
      <div class="ana-tabs">${tabs.map(([k, l]) => `<button type="button" data-k="${k}" class="${k === ANA_TAB ? 'on' : ''}" onclick="verGrafico('${k}')">${l}</button>`).join('')}</div>
    </div>
    <div class="stats">
      ${st('Ventas del mes', nuevas.length, 'desde el ' + fmtD(iniS), '#0077b6')}
      ${st('Hoy', delHoy.length, 'ayer ' + nAyer + (ultima ? ' · última ' + ultima : ''), '#0891b2')}
      ${st('Formalizadas', formal, '', '#15803d')}
      ${st('Bajas', bajas, '', '#7e22ce')}
      ${st('% de baja', pct === null ? '—' : pct.toFixed(1) + '%', 'bajas ÷ ventas', '#dc2626')}
    </div>
    <div id="ana-sub" class="small text-muted mt-2"></div>
    <div id="ana-chart" style="height:200px"></div></div>`;

}

function calcLiq(lista, cfg, comp, mes) {
  const num = x => parseFloat(x) || 0, r2 = x => Math.round(x * 100) / 100;
  const p = { luz: num(cfg.pLuz), gas: num(cfg.pGas), sve: num(cfg.pSve), svg: num(cfg.pSvg), socio: num(cfg.pSocio) };
  const b = { luz: num(cfg.bLuz), gas: num(cfg.bGas), sve: num(cfg.bSve), svg: num(cfg.bSvg) };
  const tramos = cfg.usarTramos ? (cfg.tramos || []).filter(t => t.dias !== '' && t.pct !== '')
    .map(t => ({ dias: num(t.dias), pct: Math.min(100, Math.max(0, num(t.pct))) })).sort((a, z) => a.dias - z.dias) : [];
  const [Y, M] = mes.split('-').map(Number), finS = new Date(Y, M, 0).toLocaleDateString('sv-SE');
  const del = lista.filter(c => c.compania === comp);
  const enMes = f => !!f && f.slice(0, 7) === mes;
  const q = c => ({ luz: !!c.cups_luz, gas: !!c.cups_gas, sve: !!c.cups_luz && c.sva_luz === 'SI', svg: !!c.cups_gas && c.svg === 'SI', socio: c.contrata === 'ONG' });
  const comi = c => { const u = q(c); return (u.luz ? p.luz : 0) + (u.gas ? p.gas : 0) + (u.sve ? p.sve : 0) + (u.svg ? p.svg : 0) + (u.socio ? p.socio : 0); };
  const dif = (a, z) => Math.round((new Date(z + 'T00:00:00') - new Date(a + 'T00:00:00')) / 864e5);
  const vend = c => (c.owner && (c.owner.nombre || c.owner.email)) || '';
  const fila = c => { const u = q(c); return { ref: c.referencia, cliente: c.cliente, dni: c.dni || '', cupsLuz: c.cups_luz || '', cupsGas: c.cups_gas || '', vendedor: vend(c), luz: u.luz ? 1 : 0, gas: u.gas ? 1 : 0, sve: u.sve ? 1 : 0, svg: u.svg ? 1 : 0, socio: u.socio ? 1 : 0 }; };
  const avisos = [];
  const formal = del.filter(c => c.estado === 'FORMALIZADO' && enMes(c.fecha_activacion)).map(c => ({ ...fila(c), importe: r2(comi(c)) }));
  const sinAct = del.filter(c => c.estado === 'FORMALIZADO' && !c.fecha_activacion);
  if (sinAct.length) avisos.push(sinAct.length + ' contrato(s) FORMALIZADO sin fecha de activación no se cuentan: ' + sinAct.map(c => c.referencia).join(', '));
  const sinBaja = del.filter(c => c.estado === 'BAJA' && !c.fecha_baja);
  if (sinBaja.length) avisos.push(sinBaja.length + ' contrato(s) en BAJA sin fecha de baja no se cuentan: ' + sinBaja.map(c => c.referencia).join(', '));
  const bajas = del.filter(c => c.estado === 'BAJA' && enMes(c.fecha_baja)).map(c => {
    let pct = null, desc = 0, nota = '', d = null;
    if (comp === ONG) { pct = 100; desc = comi(c); }
    else if (cfg.usarTramos) {
      if (!c.fecha_activacion) { nota = 'Sin fecha de activación'; avisos.push(c.referencia + ': baja sin fecha de activación, no se puede calcular el porcentaje y no se descuenta.'); }
      else { d = dif(c.fecha_activacion, c.fecha_baja); const t = tramos.find(x => d <= x.dias); pct = t ? t.pct : 0; desc = comi(c) * pct / 100; if (!t) nota = 'Fuera de los tramos: no descuenta'; }
    } else { const u = q(c); desc = (u.luz ? b.luz : 0) + (u.gas ? b.gas : 0) + (u.sve ? b.sve : 0) + (u.svg ? b.svg : 0); }
    return { ...fila(c), activacion: c.fecha_activacion || '', baja: c.fecha_baja, dias: d, pct, comision: r2(comi(c)), descuento: r2(desc), nota };
  });
  if (bajas.length && comp !== ONG) {
    if (cfg.usarTramos && !tramos.length) avisos.push('Activaste el porcentaje por días pero no escribiste ningún tramo: las bajas no descuentan nada.');
    if (!cfg.usarTramos && !(b.luz || b.gas || b.sve || b.svg)) avisos.push('Hay bajas pero no escribiste el valor de la baja: no se descuenta nada.');
  }
  if (!(p.luz || p.gas || p.sve || p.svg || p.socio)) avisos.push('No has escrito lo que pagan por cada producto.');
  const bnd = del.filter(c => c.estado === 'BAJA NO DESCOMISIONABLE' && enMes(c.fecha_baja)).length;
  const nf = del.filter(c => !['FORMALIZADO', 'BAJA', 'BAJA NO DESCOMISIONABLE'].includes(c.estado) && c.created_at && new Date(c.created_at).toLocaleDateString('sv-SE') <= finS);
  const porEstado = {}; nf.forEach(c => { porEstado[c.estado] = (porEstado[c.estado] || 0) + 1; });
  const sm = (arr, k) => arr.reduce((s, x) => s + x[k], 0);
  const resumen = { nFormal: formal.length, fLuz: sm(formal, 'luz'), fGas: sm(formal, 'gas'), fSve: sm(formal, 'sve'), fSvg: sm(formal, 'svg'), fSocio: sm(formal, 'socio'), importeF: r2(sm(formal, 'importe')),
    nBajas: bajas.length, bLuz: sm(bajas, 'luz'), bGas: sm(bajas, 'gas'), bSve: sm(bajas, 'sve'), bSvg: sm(bajas, 'svg'), bSocio: sm(bajas, 'socio'), descB: r2(sm(bajas, 'descuento')),
    nBnd: bnd, nNF: nf.length, porEstado };
  resumen.total = r2(resumen.importeF - resumen.descB);
  return { compania: comp, mes, cfg, resumen, formal, bajas, avisos };
}

function zipStore(files) {
  const enc = new TextEncoder(), partes = [], central = []; let off = 0;
  const u16 = n => [n & 255, (n >>> 8) & 255], u32 = n => [n & 255, (n >>> 8) & 255, (n >>> 16) & 255, (n >>> 24) & 255];
  files.forEach(([nombre, texto]) => {
    const nb = enc.encode(nombre), d = enc.encode(texto), crc = crc32(d);
    const cab = [...u32(0x04034b50), ...u16(20), ...u16(0x0800), ...u16(0), ...u16(0), ...u16(0x21), ...u32(crc), ...u32(d.length), ...u32(d.length), ...u16(nb.length), ...u16(0)];
    partes.push(new Uint8Array(cab), nb, d);
    central.push(new Uint8Array([...u32(0x02014b50), ...u16(20), ...u16(20), ...u16(0x0800), ...u16(0), ...u16(0), ...u16(0x21), ...u32(crc), ...u32(d.length), ...u32(d.length), ...u16(nb.length), ...u16(0), ...u16(0), ...u16(0), ...u16(0), ...u32(0), ...u32(off)]), nb);
    off += cab.length + nb.length + d.length;
  });
  const tamC = central.reduce((s, x) => s + x.length, 0);
  const fin = new Uint8Array([...u32(0x06054b50), ...u16(0), ...u16(0), ...u16(files.length), ...u16(files.length), ...u32(tamC), ...u32(off), ...u16(0)]);
  const todo = [...partes, ...central, fin], out = new Uint8Array(todo.reduce((s, x) => s + x.length, 0)); let p = 0;
  todo.forEach(x => { out.set(x, p); p += x.length; });
  return out;
}

/* ===== Lectura completa por tandas (Supabase entrega como máximo 1000 filas por consulta) ===== */
async function todas(construir) {
  const out = [];
  for (let i = 0; ; i += 1000) {
    const { data, error } = await construir().range(i, i + 999);
    if (error) return { data: null, error };
    out.push(...(data || []));
    if (!data || data.length < 1000) return { data: out, error: null };
  }
}

    const sd = h.filas.map((fi, r) => `<row r="${r + 1}"${fi.h ? ` ht="${fi.h}" customHeight="1"` : ''}>${(fi.celdas || []).map((c, i) => {

async function guardarTarifas() {
  const desde = document.getElementById('com-desde').value;
  if (!desde) { alert('Elige la fecha "vigente desde".'); return; }
  const valores = {};
  if (COM.compania === ONG) {
    const n = x => { const v = parseFloat(x); return isNaN(v) ? null : v; };
    const tramos = [...document.querySelectorAll('#ct-body [data-ct]')].map(r => {
      const t = { desde: n(r.querySelector('.ct-desde').value) };
      COLS.forEach(([c]) => { t[c] = n(r.querySelector('.ct-' + c).value) || 0; });
      return t;
    }).filter(t => t.desde !== null).sort((a, z) => a.desde - z.desde);
    if (!tramos.length) { alert('Escribe al menos un tramo con su "Donación desde".'); return; }
    if (new Set(tramos.map(t => t.desde)).size !== tramos.length) { alert('Hay dos tramos con la misma "Donación desde". Corrígelo antes de guardar.'); return; }
    valores.socio_tramos = tramos;
  } else {
    PRODS(COM.compania).forEach(([p]) => { valores[p] = {}; COLS.forEach(([c]) => { const x = parseFloat(document.getElementById(`t-${p}-${c}`).value); valores[p][c] = isNaN(x) ? 0 : x; }); });
  }
  const { error } = await db.from('comision_tarifas').upsert({ compania: COM.compania, vigente_desde: desde, valores }, { onConflict: 'compania,vigente_desde' });
  if (error) { alert('No se pudo guardar: ' + error.message); return; }
  aviso('Valores de ' + COM.compania + ' guardados. Se aplican a las ventas activadas desde el ' + fmtD(desde) + '.');
  await cargarTarifas(); resumenComisiones(); pintarMiComision();
}
async function borrarTarifa(id) {
  if (!confirm('¿Eliminar esta versión de valores? Las ventas de esas fechas usarán la versión anterior.')) return;
  const { error } = await db.from('comision_tarifas').delete().eq('id', id);
  if (error) { alert('No se pudo eliminar: ' + error.message); return; }
  await cargarTarifas(); resumenComisiones();
}
async function resumenComisiones() {
  const mes = document.getElementById('com-mes').value, el = document.getElementById('com-res');
  if (!mes) return;
  const { data, error } = await db.rpc('comisiones_mes', { p_mes: mes });
  if (error) { el.innerHTML = `<div class="p-3 text-danger">${esc(error.message)}</div>`; return; }
  const filas = Object.values(agruparComisiones(data)).sort((a, z) => z.neto - a.neto), fm = n => esc(fmMon(n, '€'));
  el.innerHTML = filas.length
    ? `<table class="table align-middle mb-0"><thead class="table-light"><tr><th>Vendedor</th><th class="text-end">Confirmada</th><th class="text-end">Bajas</th><th class="text-end">Neto</th><th class="text-end">Potencial</th></tr></thead><tbody>${filas.map(x => `<tr><td>${esc(x.nombre)}</td><td class="text-end">${fm(x.conf)}</td><td class="text-end text-danger">${fm(x.baja)}</td><td class="text-end fw-bold">${fm(x.neto)}</td><td class="text-end text-muted">${fm(x.pot)}</td></tr>`).join('')}</tbody></table>`
    : '<div class="p-3 text-muted">No hay comisiones en ese mes.</div>';
}

function iniVivo() {
  if (canal || !db.channel || !perfil) return;

      p => { contarNotifs(); if (p.new && p.new.texto) aviso(p.new.texto, 'info'); })
    .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'contrato_comentarios' },
      p => { const id = p.new && p.new.contrato_id; if (id && document.getElementById('chat-' + id)) listarChat(id); })
    .subscribe();
}
function programarRefresco() { clearTimeout(rtTimer); rtTimer = setTimeout(refrescarVivo, 700); }
async function refrescarVivo() {
  const { data, error } = await todas(() => db.from('contratos').select('*, owner:profiles!owner_id(nombre,email)').order('created_at', { ascending: false }).order('id', { ascending: false }));
  if (error || !data) return;
  const previo = Object.fromEntries(contratos.map(c => [c.id, c.estado]));
  contratos = data; dash(); pintarMiComision();
  const tb = document.getElementById('tbody');
  if (hayAbiertos() || (document.activeElement && tb.contains && tb.contains(document.activeElement))) {
    // hay un contrato desplegado: solo se actualizan los estados para no cerrar lo que se está viendo
    pendRender = true;
    contratos.forEach(c => {
      const eb = document.getElementById('eb-' + c.id); if (eb) eb.innerHTML = badge(c.estado);
      const se = document.getElementById('est-' + c.id); if (se && se.value === previo[c.id] && document.activeElement !== se) se.value = c.estado;
    });
  } else { pendRender = false; render(); }
}
